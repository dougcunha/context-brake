import { expect, mock, test } from 'claude-code/testing'

const CONFIG = JSON.stringify({ schemaVersion: 1, activeHarnesses: ['claude-code'], telemetry: { injectionMode: 'threshold_only', activationThresholdPercentage: 50, contextWindowCeiling: 128000, zones: { greenMaxPercentage: 49, yellowMaxPercentage: 65, criticalPercentage: 75 } }, snapshot: { triggerZone: 'RED', command: 'sdd-snapshot' }, autoRestart: { maxConsecutiveRestarts: 10 } })

test('handoff command submits its prompt', async ($, on) => {
  const clock = mock.clock(on, { now: 1_000 })
  const seen = { submits: [] as string[], logs: [] as string[], writes: [] as string[] }
  on('fs.exists', async (_, e) => ({ value: String((e as { path: string }).path).endsWith('context-brake.config.json') }) as never)
  on('fs.read', async () => ({ value: CONFIG }) as never)
  on('fs.write', async (_, e) => (seen.writes.push(String((e as { path: string }).path)), { value: undefined }) as never)
  on('store.get', async () => ({ value: undefined }) as never)
  on('store.set', async () => ({ value: undefined }) as never)
  on('session.root', async () => ({ value: 'D:/repo' }) as never)
  on('session.surfaces', async () => ({ value: ['terminal'] }) as never)
  on('ui.log', async (_, e) => (seen.logs.push(String((e as { text: string }).text)), { value: undefined }) as never)
  on('command.register', async (_, e) => ({ value: { command: (e as { name: string }).name } }) as never)
  on('prompt.submit', async (_, e) => (seen.submits.push(String((e as { text: string }).text)), { text: (e as { text: string }).text }) as never)
  on('session.start', async (_, e) => ({ cwd: (e as { cwd: string }).cwd }) as never)
  on('session.version', async () => ({ value: { version: '2.1.306' } }) as never)
  on('env.get', async () => ({ value: undefined }) as never)
  on('session.usage', async () => ({ value: { context: { window: 1000000 } } }) as never)
  await $.session.start({ cwd: 'D:/repo', surface: 'terminal', isInteractive: true } as never)
  const result = await $.command.run({ command: 'handoff', args: '' } as never)
  await clock.advance(500)
  console.log('RESULT', JSON.stringify(result))
  console.log('SUBMITS', JSON.stringify(seen.submits))
  console.log('LOGS', JSON.stringify(seen.logs))
  expect(seen.submits.length).toBe(1)
})
