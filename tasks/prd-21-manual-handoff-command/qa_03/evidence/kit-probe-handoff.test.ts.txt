import { expect, mock, test } from 'claude-code/testing'

// QA qa_03 probe: drives the installed ContextBrake mod through $.command.run with the claude plugin test kit.
const BASE = { schemaVersion: 1, activeHarnesses: ['claude-code'], telemetry: { injectionMode: 'threshold_only', activationThresholdPercentage: 50, contextWindowCeiling: 128000, zones: { greenMaxPercentage: 49, yellowMaxPercentage: 65, criticalPercentage: 75 } } }
const SNAPSHOT_CONFIG = { ...BASE, snapshot: { triggerZone: 'RED', command: 'sdd-snapshot' }, autoRestart: { maxConsecutiveRestarts: 10 } }
const HANDOFF_CONFIG = { ...BASE, snapshot: { triggerZone: 'RED' }, autoRestart: { maxConsecutiveRestarts: 10 } }
const OFF_CONFIG = { ...BASE, snapshot: { triggerZone: 'RED', command: 'sdd-snapshot' } }
const LEAD = 'ContextBrake: the user typed /handoff to continue this work in a new session. Do this now:'
const ACCEPTED = 'ContextBrake: asking the agent to save its state; the session restarts when the reply ends.'
const OFF_LINE = 'ContextBrake: /handoff needs automatic restart. Run npx context-brake init --auto-restart and start a new session.'
const GUARD_KEY = 'contextbrake:autorestart:D:/repo'

type Seen = { submits: string[]; submitTimes: number[]; logs: string[]; writes: { path: string; text: string }[]; store: Record<string, unknown>; sets: unknown[] }

function scene(on: Parameters<Parameters<typeof test>[1]>[1], config: object, env: Record<string, string> = {}): { seen: Seen; clock: ReturnType<typeof mock.clock> } {
  const clock = mock.clock(on, { now: 1_000 })
  const seen: Seen = { submits: [], submitTimes: [], logs: [], writes: [], store: { [GUARD_KEY]: { consecutive: 2, toolCallsSinceSeed: null } }, sets: [] }
  const files: Record<string, string> = {}
  on('fs.exists', async (_, e) => ({ value: String((e as { path: string }).path).endsWith('context-brake.config.json') || files[String((e as { path: string }).path)] !== undefined }) as never)
  on('fs.read', async (_, e) => { const p = String((e as { path: string }).path); return ({ value: p.endsWith('context-brake.config.json') ? JSON.stringify(config) : files[p] ?? '' }) as never })
  on('fs.write', async (_, e) => { const w = e as { path: string; text?: string; content?: string; data?: string }; const text = String(w.text ?? w.content ?? w.data ?? ''); files[String(w.path)] = text; seen.writes.push({ path: String(w.path), text }); return ({ value: undefined }) as never })
  on('fs.stat', async () => ({ value: undefined }) as never)
  on('store.get', async (_, e) => ({ value: seen.store[(e as { key: string }).key] }) as never)
  on('store.set', async (_, e) => { const s = e as { key: string; value: unknown }; seen.store[s.key] = s.value; seen.sets.push(s.value); return ({ value: undefined }) as never })
  on('session.root', async () => ({ value: 'D:/repo' }) as never)
  on('session.id', async () => ({ value: 'session-1' }) as never)
  on('session.surfaces', async () => ({ value: ['terminal'] }) as never)
  on('ui.log', async (_, e) => (seen.logs.push(String((e as { text: string }).text)), { value: undefined }) as never)
  on('command.register', async (_, e) => ({ value: { command: (e as { name: string }).name } }) as never)
  on('prompt.submit', async (_, e) => (seen.submits.push(String((e as { text: string }).text)), seen.submitTimes.push(Date.now()), { text: (e as { text: string }).text }) as never)
  on('session.start', async (_, e) => ({ cwd: (e as { cwd: string }).cwd }) as never)
  on('session.version', async () => ({ value: { version: '2.1.306' } }) as never)
  on('env.get', async (_, e) => ({ value: env[String((e as { name?: string; key?: string }).name ?? (e as { key?: string }).key)] }) as never)
  on('session.usage', async () => ({ value: { context: { window: 1000000 } } }) as never)
  return { seen, clock }
}

async function run($: never, args: string): Promise<{ text?: string }> {
  const d = $ as unknown as { session: { start: (e: unknown) => Promise<unknown> }; command: { run: (e: unknown) => Promise<{ text?: string }> } }
  await d.session.start({ cwd: 'D:/repo', surface: 'terminal', isInteractive: true })
  return d.command.run({ command: 'handoff', args })
}

function report(name: string, result: unknown, seen: Seen, extra: Record<string, unknown> = {}): void {
  console.log(`[${name}]`, JSON.stringify({ result, submits: seen.submits, logs: seen.logs, writes: seen.writes.map((w) => w.path), guardSets: seen.sets, ...extra }))
}

test('P1 snapshot mode: accepted line, prompt reaches prompt.submit only after the hook returns', async ($, on) => {
  const { seen, clock } = scene(on, SNAPSHOT_CONFIG)
  const result = await run($ as never, '')
  const atReturn = seen.submits.length
  await clock.advance(50)
  const at50 = seen.submits.length
  await clock.advance(450)
  report('P1', result, seen, { atReturn, at50 })
  expect(result.text).toBe(ACCEPTED)
  expect(atReturn).toBe(0)
  expect(seen.submits).toEqual([`${LEAD} run "/sdd-snapshot", then end reply with [REQUEST_SESSION_RESET].`])
  expect(seen.logs).toEqual([])
  expect(seen.store[GUARD_KEY]).toEqual({ consecutive: 0, toolCallsSinceSeed: null })
  expect(seen.writes.some((w) => w.text.includes('ERROR_INTERNAL'))).toBe(false)
})

test('P2 focus: normalized focus in the prompt, never in a written file', async ($, on) => {
  const { seen, clock } = scene(on, SNAPSHOT_CONFIG)
  const result = await run($ as never, '  the parser\n  bug QA3MARK  ')
  await clock.advance(500)
  report('P2', result, seen)
  expect(result.text).toBe(ACCEPTED)
  expect(seen.submits).toEqual([`${LEAD} run "/sdd-snapshot", then end reply with [REQUEST_SESSION_RESET]. In the snapshot, stress this focus for the next session: "the parser bug QA3MARK".`])
  expect(seen.writes.some((w) => w.text.includes('QA3MARK') || w.text.includes('/handoff'))).toBe(false)
})

test('P3 handoff mode: markdown handoff prompt', async ($, on) => {
  const { seen, clock } = scene(on, HANDOFF_CONFIG)
  const result = await run($ as never, 'x'.repeat(600))
  await clock.advance(500)
  report('P3', result, seen, { focusLength: (seen.submits[0] ?? '').match(/"(x+)"/)?.[1]?.length })
  expect(result.text).toBe(ACCEPTED)
  expect(seen.submits[0]).toBe(`${LEAD} save handoff to .context-brake/handoff.md, end reply with [REQUEST_SESSION_RESET]. In the handoff, stress this focus for the next session: "${'x'.repeat(500)}".`)
})

test('P4 no autoRestart: off line, no prompt, counter unchanged', async ($, on) => {
  const { seen, clock } = scene(on, OFF_CONFIG)
  const result = await run($ as never, '')
  await clock.advance(500)
  report('P4', result, seen)
  expect(result.text).toBe(OFF_LINE)
  expect(seen.submits).toEqual([])
  expect(seen.store[GUARD_KEY]).toEqual({ consecutive: 2, toolCallsSinceSeed: null })
})

test('P5 kill switch: SKIP_DISABLED_ENV notice, code logged, no prompt, counter unchanged', async ($, on) => {
  const { seen, clock } = scene(on, SNAPSHOT_CONFIG, { CONTEXT_BRAKE_AUTO_RESTART: '0' })
  const result = await run($ as never, '')
  await clock.advance(500)
  report('P5', result, seen, { logText: seen.writes.filter((w) => w.path.includes('restart')).map((w) => w.text) })
  expect(result.text).toMatch(/^ContextBrake: /)
  expect(result.text).not.toBe(ACCEPTED)
  expect(seen.submits).toEqual([])
  expect(seen.writes.some((w) => w.text.includes('SKIP_DISABLED_ENV'))).toBe(true)
  expect(seen.logs).toEqual([])
  expect(seen.store[GUARD_KEY]).toEqual({ consecutive: 2, toolCallsSinceSeed: null })
})
